# Accepted lint findings, read by run-lint.sh.
#
# One waiver per line:  <phase> <regex>
#   <phase>  verilator | yosys | slang | verible
#   <regex>  extended regex (grep -E), matched against one finding line as it
#            appears in <phase>.all of a run-lint.sh work directory
# A finding that matches is waived. Put the reason on a comment line above it.
#
# Guideline: resolve a finding at its root; waive only as the last resort, by
# the same steps as openPCIE's run-lint-waivers.txt:
#   1. dead logic in our RTL: delete it;
#   2. oversized buses in our RTL: size them to what is used;
#   3. unused outputs of hard macros, primitives and third-party cores:
#      .PIN (/*LINT:OK*/), or route them to an NC_ (no-connect) signal;
#   4. only then a waiver here, with its reason.

# NC_ signals are no-connects by name: driven by an unused output, never read
verilator ^%Warning-UNUSEDSIGNAL: .*'NC_[A-Za-z0-9_]+'

# KEEPME: the SOC's rvfi (the CPU's retired instructions) is read only by
# WaveCrux's RISC-V views (tb_soc, 5.sim) and formal checks, never by the RTL
verilator ^%Warning-UNUSEDSIGNAL: 2\.rtl/6\.soc/rvfi_if\.sv:[0-9]+:[0-9]+: Signal is not used: 'rvfi_[a-z0-9_]+'$

# The SerDes BFM is a behavioural model, not RTL: its # delays are computed
verilator ^%Warning-ZERODLY: 2\.rtl/1\.phy/2\.serdes-model/

# Generated by PeakRDL: left as the generator writes it (as on openPCIE)
verilator ^%Warning-UNUSEDSIGNAL: csr/csr\.sv:[0-9]+:[0-9]+: Signal is not used: '(decoded_err|is_valid_addr|is_valid_rw)'
verilator ^%Warning-MULTIDRIVEN: csr/csr\.sv:[0-9]+:[0-9]+: Variable written to in always_(comb|ff) also written by other process .*: '(field_combo|field_storage)'$

# The TL (pcie_tl) is an empty box: it ties its outputs off and reads none of
# its inputs, so the DLL's receive side, the SOC's TLP stream and the TL's
# own clock and reset are unread, and the pcie_pkg constants for the TL are
# unused
verilator ^%Warning-UNUSEDSIGNAL: 2\.rtl/4\.dll/dll_if\.sv:[0-9]+:[0-9]+: Signal is not used: '(tx_tlp_ready|rx_[a-z_]+)'$
verilator ^%Warning-UNUSEDSIGNAL: 2\.rtl/5\.tlp/axis_if\.sv:[0-9]+:[0-9]+: Signal is not used: '(tdata|tkeep|tuser|tlast|tvalid)'$
verilator ^%Warning-UNUSEDSIGNAL: 2\.rtl/5\.tlp/pcie_tl\.sv:[0-9]+:[0-9]+: Signal is not used: '(clk|rst_n)'$
verilator ^%Warning-UNUSEDPARAM: 2\.rtl/5\.tlp/pcie_pkg\.sv:[0-9]+:[0-9]+: Parameter is not used: '(TRN_KEEP_W|TRN_REM_W|TLP_TYPE_CPL|TLP_TYPE_CFG)'$

# Cologne Chip submodule code (2.pipe/pipe_src) keeps its own style and is not
# changed here: its style findings are waived. Findings that can be bugs
# (LATCH, undriven wires) are not: they go upstream to gatemate-pipe.
# These waivers are temporary. Closer to the first release, the Cologne Chip
# code is brought up to our coding standard and runs through this QA flow,
# and the waivers go.
verilator ^%Warning-(GENUNNAMED|WIDTHEXPAND|WIDTHTRUNC|UNUSEDSIGNAL|UNUSEDPARAM|CASEINCOMPLETE|IMPLICIT): 2\.rtl/2\.pipe/pipe_src/
